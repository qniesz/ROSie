# Requirements for Pi Zero 2 W (arm64, no Docker, no ROS 2)
#
# Install with:
#   pip install -r zero_requirements.txt
#
# Or use distro packages for heavy compiled libs (faster on Pi Zero):
#   sudo apt install python3-numpy python3-scipy python3-pil
#   pip install -r zero_requirements.txt --no-deps   (skip numpy/scipy/Pillow)
#
# Notes:
#   - paho-mqtt MUST stay <2.0 (breaking API changes in v2).
#   - breezyslam is compiled from C; needs gcc + python3-dev.
#     On Pi OS Bookworm: sudo apt install gcc python3-dev  (already present).
#   - psutil is optional; diagnostics are skipped if not installed.

# --- Serial / MQTT (same as Pi 4 driver) ---
pyserial>=3.5
paho-mqtt>=1.6,<2.0
RPi.GPIO>=0.7

# --- Map processing (clean_map.py) ---
numpy>=1.24
scipy>=1.10
Pillow>=9.4

# --- BreezySLAM (replaces SLAM Toolbox) ---
# Pure-C core wrapped in Python; compiles in ~2 min on Pi Zero 2 W.
breezyslam>=0.9.1

# --- Diagnostics (optional but recommended) ---
psutil>=5.9
